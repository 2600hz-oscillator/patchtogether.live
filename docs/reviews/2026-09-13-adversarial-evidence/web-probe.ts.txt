it.fails('one heartbeat remains one sample despite cursor traffic', () => {
  const doc = new Y.Doc();
  const aw = new Awareness(doc);
  let now = 1000;
  aw.setLocalStateField('__heartbeat', { tick: 1, ts_ms: 10000 });
  const clock = createSharedClock({ provider: { awareness: aw } as never, ydoc: doc,
    deps: { perfNow: () => now, setInterval: (() => 0) as never, clearInterval: () => {} } });
  const initial = { ...clock.snapshot };
  for (let i = 1; i <= 10; i++) {
    now = 1000 + i * 50;
    aw.setLocalStateField('cursor', { x: i, y: 0 });
  }
  const final = { ...clock.snapshot };
  console.log('PROBE clock_cursor', JSON.stringify({ initial, final, resyncCount: clock.resyncCount }));
  clock.destroy(); aw.destroy(); doc.destroy();
  expect(final.sampleCount).toBe(1);
});

  const rows=[];

    const patch=syncedStore({nodes:{},edges:{}}) as any; const doc=getYjsDoc(patch);
    doc.transact(()=>{for(let i=0;i<size;i++){
      patch.nodes['n'+i]={id:'n'+i,type:'analogVco',domain:'audio',position:{x:0,y:0},params:{gain:0}};
      if(i>0)patch.edges['e'+i]={id:'e'+i,source:{nodeId:'n'+(i-1),portId:'out'},target:{nodeId:'n'+i,portId:'in'},sourceType:'audio',targetType:'audio'};
    }});
    const bus=createSnapshotBus({patch,ydoc:doc}); let emissions=0;
    const off=bus.subscribe(()=>{emissions++;});
    for(let i=0;i<20;i++)doc.getMap('layouts').set('review',i);
    emissions=0; const begin=performance.now();
    for(let i=0;i<300;i++)doc.getMap('layouts').set('review',i+100);
    const layoutMs=performance.now()-begin;
    const beginParam=performance.now();
    for(let i=0;i<300;i++)patch.nodes.n0.params.gain=i;
